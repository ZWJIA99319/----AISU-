cmake_minimum_required(VERSION 3.8)
project(vehicle_bridge)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs serial)
 find_package(${dep} REQUIRED)
endforeach()
add_executable(vehicle_bridge_node src/bridge.cpp src/frame_codec.cpp)
target_include_directories(vehicle_bridge_node PRIVATE include)
ament_target_dependencies(vehicle_bridge_node rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs serial)
install(TARGETS vehicle_bridge_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
